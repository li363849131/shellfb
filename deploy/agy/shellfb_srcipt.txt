#!/bin/bash
# ==========================================================
# 终极 Web AI 编程工作站 - 一键部署归档脚本 (2026 完整全功能版)
# 核心组件: ShellFromBrowser (包含最新 DOCX 1:1预览/XLSX Sheet标签/文本编辑/文件删除/全局检索/Ctrl+V上传/断点续传/chafa高清/低清预览)
#           FileBrowser网盘 + Tmux多Session同步 + Cloudflared穿透 + AGY/Claude/Codex
# 终端端口: 30009 | 网盘端口: 30010 | 默认账号: admin | 默认密码: gjp
# ==========================================================

echo -e "\n\033[1;36m[1/9] 正在清理旧环境与残留...\033[0m"
systemctl stop shellfb cf-tunnel filebrowser cf-filebrowser 2>/dev/null
systemctl disable shellfb cf-tunnel filebrowser cf-filebrowser 2>/dev/null
rm -rf /etc/shellfb /etc/filebrowser /etc/systemd/system/shellfb.service /etc/systemd/system/cf-tunnel.service /etc/systemd/system/filebrowser.service /etc/systemd/system/cf-filebrowser.service
apt-get remove -y golang-go golang-* 2>/dev/null
apt-get autoremove -y 2>/dev/null

echo -e "\n\033[1;36m[2/9] 安装基础系统依赖 (含图形预览 chafa、文档转换 libreoffice、剪贴板与 Tmux 支持)...\033[0m"
export DEBIAN_FRONTEND=noninteractive
apt update && apt install -y curl wget git unzip openssl python3-bcrypt xclip wl-clipboard tmux chafa libreoffice

# 配置 tmux
cat > /root/.tmux.conf << 'EOF'
set -g mouse off
set -g allow-rename off
set -g history-limit 50000
set -g default-terminal "screen-256color"
set -g base-index 1
setw -g pane-base-index 1
set -g renumber-windows on
setw -g aggressive-resize on
EOF

# 写入 setname 重命名别名函数
cat >> /root/.bashrc << 'EOF'
setname() {
    if [ -z "$1" ]; then
        echo "使用方法: setname <项目或Session名称>"
        return 1
    fi
    export SESSION_TITLE="$1"
    if [ -n "$TMUX" ]; then
        tmux rename-window "$1" 2>/dev/null
    fi
    echo -e "\033[1;32m✅ 当前终端窗口已成功重命名为: \033[1;33m$1\033[0m"
}
alias rename='setname'

_update_prompt() {
    local TITLE_PREFIX=""
    if [ -n "$SESSION_TITLE" ]; then
        TITLE_PREFIX="\[\033[1;33m\]📌【$SESSION_TITLE】\[\033[0m\] "
    fi
    PS1="${TITLE_PREFIX}\[\033[1;32m\]\u@\h\[\033[0m\]:\[\033[1;34m\]\w\[\033[0m\]\$ "
}
if [[ ";$PROMPT_COMMAND;" != *"_update_prompt;"* ]]; then
    PROMPT_COMMAND="_update_prompt${PROMPT_COMMAND:+;$PROMPT_COMMAND}"
fi
EOF

# 创建多 Session 智能绑定脚本
cat > /usr/local/bin/tmux-web-shell << 'EOF'
#!/bin/bash
export PATH="/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin:/root/.local/bin:$PATH"
export TERM="xterm-256color"

if [ -n "$TMUX" ]; then
    exec /bin/bash
fi

MY_PID=$$
SHELLFB_PID=$(pgrep -x shellfb | head -n 1)

INDEX=1
if [ -n "$SHELLFB_PID" ]; then
    PIDS=$(pgrep -P "$SHELLFB_PID" -f tmux-web-shell | sort -n)
    CALC_IDX=1
    for pid in $PIDS; do
        if [ "$pid" -eq "$MY_PID" ]; then
            INDEX=$CALC_IDX
            break
        fi
        CALC_IDX=$((CALC_IDX + 1))
    done
fi

SESS_NAME="session${INDEX}"
if tmux has-session -t "$SESS_NAME" 2>/dev/null; then
    exec tmux attach-session -t "$SESS_NAME"
else
    exec tmux new-session -s "$SESS_NAME"
fi
EOF
chmod +x /usr/local/bin/tmux-web-shell

echo -e "\n\033[1;36m[3/9] 下载并配置官方最新版 Go 环境...\033[0m"
LATEST_GO=$(curl -s https://go.dev/VERSION?m=text | head -n1)
wget -q https://go.dev/dl/${LATEST_GO}.linux-amd64.tar.gz
rm -rf /usr/local/go
tar -C /usr/local -xzf ${LATEST_GO}.linux-amd64.tar.gz
rm -f ${LATEST_GO}.linux-amd64.tar.gz
export PATH=/usr/local/go/bin:$PATH

echo -e "\n\033[1;36m[4/9] 编译并安装 Web 终端 (ShellFromBrowser)...\033[0m"
export GOPATH=$HOME/go
export PATH=$PATH:$GOPATH/bin

# 如果 /root/shellfb-src 目录下包含源码则直接编译，否则检查当前同目录压缩包
if [ -f /root/agy/shellfb-src.tar.gz ] && [ ! -d /root/shellfb-src ]; then
    tar -xzf /root/agy/shellfb-src.tar.gz -C /root
fi

cd /root/shellfb-src && /usr/local/go/bin/go build -o /usr/local/bin/shellfb ./cmd/shellfb
cd /root

# 生成配置
mkdir -p /etc/shellfb /root/transfers
JWT_SECRET=$(openssl rand -hex 32)
HASH=$(python3 -c 'import bcrypt; print(bcrypt.hashpw(b"gjp", bcrypt.gensalt()).decode())')

cat > /etc/shellfb/config.yaml << EOF
server:
  addr: "127.0.0.1:30009"
  tls:
    enabled: false
auth:
  enabled: true
  jwt_secret: "${JWT_SECRET}"
  users:
    - username: admin
      password_hash: "${HASH}"
shell:
  command: "/bin/bash"
  env:
    - "TERM=xterm-256color"
sessions:
  max_per_user: 20
  idle_timeout: "720h"
transfer:
  dir: "/root"
EOF

# 写入终端服务
cat > /etc/systemd/system/shellfb.service << 'EOF'
[Unit]
Description=ShellFromBrowser Web Terminal
After=network.target

[Service]
Environment="PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin:/root/.local/bin"
ExecStart=/usr/local/bin/shellfb --config /etc/shellfb/config.yaml
Restart=always
RestartSec=5
User=root
WorkingDirectory=/root

[Install]
WantedBy=multi-user.target
EOF

echo -e "\n\033[1;36m[5/9] 安装并配置可视化 Web 网盘 (FileBrowser)...\033[0m"
wget -q https://github.com/filebrowser/filebrowser/releases/download/v2.32.0/linux-amd64-filebrowser.tar.gz
tar -xzf linux-amd64-filebrowser.tar.gz && mv filebrowser /usr/local/bin/ && rm -f linux-amd64-filebrowser.tar.gz LICENSE README.md
mkdir -p /etc/filebrowser
filebrowser config init -a "127.0.0.1" -p 30010 -r "/root" -d "/etc/filebrowser/filebrowser.db" > /dev/null 2>&1
filebrowser users add admin gjp --perm.admin -d "/etc/filebrowser/filebrowser.db" > /dev/null 2>&1

cat > /etc/systemd/system/filebrowser.service << 'EOF'
[Unit]
Description=FileBrowser Web File Manager
After=network.target

[Service]
ExecStart=/usr/local/bin/filebrowser -d /etc/filebrowser/filebrowser.db
Restart=always
RestartSec=5
User=root
WorkingDirectory=/root

[Install]
WantedBy=multi-user.target
EOF

echo -e "\n\033[1;36m[6/9] 安装并配置 Cloudflare 官方穿透隧道...\033[0m"
curl -L -o cloudflared.deb https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-linux-amd64.deb
dpkg -i cloudflared.deb > /dev/null 2>&1
rm -f cloudflared.deb

# Web 终端隧道
cat > /etc/systemd/system/cf-tunnel.service << 'EOF'
[Unit]
Description=Cloudflare Quick Tunnel for ShellFB
After=network.target

[Service]
ExecStart=/usr/bin/cloudflared tunnel --no-autoupdate --url http://127.0.0.1:30009
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
EOF

# FileBrowser 网盘隧道
cat > /etc/systemd/system/cf-filebrowser.service << 'EOF'
[Unit]
Description=Cloudflare Quick Tunnel for FileBrowser
After=network.target

[Service]
ExecStart=/usr/bin/cloudflared tunnel --no-autoupdate --url http://127.0.0.1:30010
Restart=always
RestartSec=5

[Install]
WantedBy=multi-user.target
EOF

echo -e "\n\033[1;36m[7/9] 部署 AI 编程神器与 Node.js 运行环境...\033[0m"
# 安装 Node.js 22.x
curl -fsSL https://deb.nodesource.com/setup_22.x | bash - > /dev/null 2>&1
apt install -y nodejs > /dev/null 2>&1

# 1. 安装 Google Antigravity
curl -fsSL https://antigravity.google/cli/install.sh | bash > /dev/null 2>&1 || true
# 2. 安装 Anthropic Claude Code
npm install -g @anthropic-ai/claude-code > /dev/null 2>&1 || true
# 3. 安装 OpenAI Codex CLI
curl -fsSL https://chatgpt.com/codex/install.sh | sh > /dev/null 2>&1 || true

echo -e "\n\033[1;36m[8/9] 强制全局注册命令，确保网页终端 100% 秒开...\033[0m"
ln -sf /root/.local/bin/agy /usr/bin/agy 2>/dev/null || true
ln -sf /root/.local/bin/codex /usr/bin/codex 2>/dev/null || true
ln -sf $(which claude 2>/dev/null) /usr/bin/claude 2>/dev/null || true

echo -e "\n\033[1;36m[9/9] 正在拉起所有服务，并生成安全域名...\033[0m"
systemctl daemon-reload
systemctl enable --now shellfb cf-tunnel filebrowser cf-filebrowser

echo -e "\n============================================="
echo "✅ 部署已全部完成！正在向 Cloudflare 申请小绿锁域名 (约需 12 秒)..."
sleep 12
URL=$(journalctl -u cf-tunnel -n 50 | grep -o 'https://.*\.trycloudflare\.com' | head -n 1)
FB_URL=$(journalctl -u cf-filebrowser -n 50 | grep -o 'https://.*\.trycloudflare\.com' | head -n 1)

echo -e "🔗 Web终端专属域名: \033[1;32m$URL\033[0m"
echo -e "📁 Web网盘专属域名: \033[1;32m$FB_URL\033[0m"
echo -e "🔑 登录账号: \033[1;32madmin\033[0m"
echo -e "🔑 登录密码: \033[1;32mgjp\033[0m"
echo -e "\n🤖 三大 AI 助手已全部就绪:"
echo -e "   - \033[1;33magy\033[0m    (Google Antigravity)"
echo -e "   - \033[1;33mclaude\033[0m (Anthropic Claude Code)"
echo -e "   - \033[1;33mcodex\033[0m  (OpenAI Codex CLI)"
echo "============================================="
